% =====================================================================
%  Appendix banner.
%  \appendix only switches the section counter to A, B, C, ...; it prints
%  nothing. \AppendixHeader opens the appendix block with an explicit
%  unnumbered "Appendix" title, so the reader sees where the main text
%  ends. \phantomsection makes hyperref anchor the \label here rather
%  than on the preceding section.
% =====================================================================
\newcommand{\AppendixHeader}{%
  \clearpage
  \phantomsection
  \label{sec:appendix}%
  \begin{center}
    \rule{\linewidth}{0.4pt}\\[0.6ex]
    {\LARGE\bfseries Appendix}\\[0.4ex]
    \rule{\linewidth}{0.4pt}
  \end{center}
  \vspace{1ex}%
}
