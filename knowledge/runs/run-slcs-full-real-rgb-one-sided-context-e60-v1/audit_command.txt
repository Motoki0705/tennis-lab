def audit(state):
    counts={'floating_tensors':0,'floating_values':0}; bad=[]
    def visit(value,path):
        if isinstance(value,torch.Tensor) and (value.is_floating_point() or value.is_complex()):
            counts['floating_tensors']+=1; counts['floating_values']+=value.numel()
            if not torch.isfinite(value).all(): bad.append({'path':path,'value':float(value) if value.numel()==1 else 'non-scalar'})
        elif isinstance(value,dict):
            for k,v in value.items(): visit(v,path+'/'+str(k))
        elif isinstance(value,(list,tuple)):
            for i,v in enumerate(value): visit(v,path+'/'+str(i))
        elif isinstance(value,float) and not math.isfinite(value): bad.append({'path':path,'value':value})
    visit(state,'')
    assert len(bad)==1 and "/callbacks/ModelCheckpoint{'monitor': None" in bad[0]['path'] and bad[0]['path'].endswith('/kth_value') and bad[0]['value']==float('inf'), bad
    return {**counts,'only_nonfinite':{'path':bad[0]['path'],'value':'positive infinity','reason':'unmonitored last-checkpoint saver kth_value sentinel'}}
